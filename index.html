Tips Permanen di GitHub Pages:

Agar semua HP perawat otomatis terhubung tanpa perlu menempel kunci satu per satu, masukkan Project URL dan Anon Key langsung ke konstanta SUPABASE_URL dan SUPABASE_ANON_KEY di baris awal <script> pada file GitHub Anda.

Script SQL Setup Tabel Supabase

Salin & jalankan sekali di menu SQL Editor Supabase Anda.

-- 1. Tabel Perawat
CREATE TABLE IF NOT EXISTS public.perawat (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  username text UNIQUE NOT NULL,
  password text NOT NULL,
  nama text NOT NULL,
  jabatan text DEFAULT 'Perawat Pelaksana',
  created_at timestamptz DEFAULT now()
);

-- 2. Tabel Pasien (Obat, Antibiotik, Transfusi, Tindakan dalam JSONB)
CREATE TABLE IF NOT EXISTS public.pasien (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  nama text NOT NULL,
  no_rm text DEFAULT '',
  bed text NOT NULL,
  kelas text DEFAULT 'Kelas 1',
  dpjp text DEFAULT '',
  diagnosa text DEFAULT '',
  bb numeric DEFAULT 60,
  obat jsonb DEFAULT '[]'::jsonb,
  antibiotik jsonb DEFAULT '[]'::jsonb,
  transfusi jsonb DEFAULT '[]'::jsonb,
  tindakan jsonb DEFAULT '[]'::jsonb,
  updated_at timestamptz DEFAULT now(),
  created_at timestamptz DEFAULT now()
);

-- 3. Tabel Balance Cairan
CREATE TABLE IF NOT EXISTS public.balance_cairan (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  pasien_id uuid REFERENCES public.pasien(id) ON DELETE CASCADE,
  nama_pasien text NOT NULL,
  bed text DEFAULT '',
  tanggal text NOT NULL,
  shift text NOT NULL,
  sisa_infus numeric DEFAULT 0,
  infus_masuk numeric DEFAULT 0,
  makan_minum numeric DEFAULT 0,
  urin numeric DEFAULT 0,
  iwl numeric DEFAULT 0,
  total_masuk numeric DEFAULT 0,
  total_keluar numeric DEFAULT 0,
  balance numeric DEFAULT 0,
  perawat text DEFAULT '',
  catatan text DEFAULT '',
  created_at timestamptz DEFAULT now()
);

-- 4. Tabel Vasoaktif (Vascon & Dobutamin)
CREATE TABLE IF NOT EXISTS public.vasoaktif (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  pasien_id uuid REFERENCES public.pasien(id) ON DELETE CASCADE,
  nama_pasien text NOT NULL,
  bed text DEFAULT '',
  jenis_obat text NOT NULL,
  bb numeric DEFAULT 60,
  sediaan_mg numeric DEFAULT 4,
  pengencer_ml numeric DEFAULT 50,
  dosis_aktif numeric DEFAULT 0.05,
  kecepatan_ml_jam numeric DEFAULT 0,
  perawat text DEFAULT '',
  updated_at timestamptz DEFAULT now()
);

-- Nonaktifkan RLS agar langsung bisa baca/tulis dari aplikasi ruangan
ALTER TABLE public.perawat DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.pasien DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.balance_cairan DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.vasoaktif DISABLE ROW LEVEL SECURITY;

-- Aktifkan Realtime Sync untuk sinkronisasi otomatis antar HP perawat
ALTER PUBLICATION supabase_realtime ADD TABLE public.pasien;
ALTER PUBLICATION supabase_realtime ADD TABLE public.balance_cairan;
ALTER PUBLICATION supabase_realtime ADD TABLE public.vasoaktif;

-- Masukkan 2 akun perawat bawaan
INSERT INTO public.perawat (username, password, nama, jabatan) VALUES
  ('perawat1', '123456', 'Ns. Rina, S.Kep', 'Perawat Pelaksana'),
  ('karu', '123456', 'Ns. Taslim Nur, S.Kep', 'Kepala Ruangan')
ON CONFLICT (username) DO NOTHING;
